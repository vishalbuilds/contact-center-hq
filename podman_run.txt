===== BUILD =====
podman build --no-cache -t contact-center-hq e:\contact-center-hq


===== RUN (no limits) =====
podman run -d --name contact-center-hq -p 8000:8000 -v C:\Users\visiv\.aws:/home/appuser/.aws -e AWS_REGION=us-east-1 contact-center-hq


===== RUN WITH RESOURCE LIMITS =====

-- Small (0.5 CPU / 256MB) --

-- Medium (1 CPU / 512MB) --
podman rm -f contact-center-hq
podman run -d --name contact-center-hq --cpus 1.0 --memory 512m -p 8000:8000 -v C:\Users\visiv\.aws:/home/appuser/.aws -e AWS_REGION=us-east-1 contact-center-hq

-- Large (2 CPU / 1GB) --
podman rm -f contact-center-hq
